name: Deploy Kin waitlist to GitHub Pages

on:
  push:
    branches: [arena/53e41efe-supoclip, main]
    paths:
      - "kin-waitlist/**"
      - ".github/workflows/kin-waitlist-pages.yml"
  workflow_dispatch:

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: pages-kin-waitlist
  cancel-in-progress: true

jobs:
  deploy:
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - uses: actions/checkout@v4
      - uses: actions/configure-pages@v5
      - uses: actions/upload-pages-artifact@v3
        with:
          path: kin-waitlist
      - id: deployment
        uses: actions/deploy-pages@v4
